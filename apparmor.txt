#include <tunables/global>

# Profil AppArmor de ClapTrap.
# Livre en mode "complain" (acces journalises, jamais bloques), comme le
# recommande la documentation Home Assistant pour un nouveau profil. Acces
# releves sur une installation reelle (6.92, 26 demarrages et toutes les
# fonctions) : ajoutes ci-dessous en 6.93 ; mode strict ensuite si une
# nouvelle mesure est vide (procedure : DOCS.md, « Securite (AppArmor) »).
# Le Supervisor charge ce profil sous le nom <depot>_claptrap.
# Pas de regle `file,` generique : elle autorisait tout acces fichier et
# rendait les regles ci-dessous decoratives, meme sans "complain".
profile claptrap flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>

  # Capacites
  signal (send) set=(kill,term,int,hup,cont),
  # Signaux d'arret (Supervisor, s6, Gunicorn) recus par les processus confines
  signal (receive) peer=unconfined,
  # Le Supervisor renomme le profil (<depot>_claptrap) : motif, pas le nom exact.
  signal (send, receive) peer=*_claptrap,
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,

  # Reseau : MQTT, API du Supervisor, flux RTSP, VBAN (UDP), webhooks,
  # interface web (ingress), socket PulseAudio
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network unix stream,
  network unix dgram,
  network netlink raw,

  # S6-Overlay (r : scripts lus par leur interpreteur, comme with-contenv)
  /init rix,
  /bin/** rix,
  /usr/bin/** rix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** rix,
  /command/** rix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  / r,

  # Fuseau horaire au demarrage (script base-app-timezone de l'image de
  # base) : lien /etc/localtime (`ln -sf` de coreutils passe par un nom
  # temporaire /etc/CuXXXXXX ; postinst de tzdata par .dpkg-new), puis
  # /etc/timezone et dpkg-reconfigure tzdata (dpkg, debconf). A reverifier
  # a chaque changement d'image de base.
  /etc/localtime rw,
  /etc/localtime.dpkg-new rw,
  /etc/timezone rw,
  /etc/Cu?????? rw,
  /usr/sbin/dpkg-reconfigure rix,
  /var/{,cache/} r,
  /var/lib/dpkg/{,**} r,
  /var/lib/dpkg/info/* ix,
  /var/cache/debconf/{,**} rwk,

  # Bashio et script de demarrage
  /usr/lib/bashio/** ix,
  /run.sh rix,
  /tmp/** rwk,

  # Application : Python (venv), Gunicorn, ffmpeg, parecord et pactl
  /usr/src/app/{,**} r,
  /usr/src/app/venv/** mr,
  /usr/src/app/venv/bin/** ix,
  /usr/local/bin/ffmpeg ix,          # ffmpeg minimal compile dans l'image
  /usr/lib/** mr,
  /usr/local/** mr,
  /usr/share/** r,
  /etc/** r,
  /proc/** r,
  /sys/** r,
  /dev/shm/{,**} rw,
  /root/.cache/** rwk,
  /root/.config/ rw,
  /root/.config/** rwk,

  # Reglages (/data/settings.json) et options de l'add-on (/data/options.json)
  /data/ r,                          # dossier lui-meme : fsync apres chaque ecriture
  /data/** rw,

  # TLS (broker MQTT externe, flux rtsps://) : certificats racine
  /etc/ssl/** r,
  /usr/share/ca-certificates/** r,

  # Audio (PulseAudio du Supervisor)
  /run/audio/** rw,
  /run/pulse/** rw,
  /dev/snd/** rw,
  /etc/pulse/** r,
}
